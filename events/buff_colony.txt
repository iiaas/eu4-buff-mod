namespace = buff_colony

# Fired from on_colony_finished for the player's finished colony (event_target:buff_finished_colony):
# every empty neighbouring province gets a Siberian frontier for ROOT (same effect as the
# "establish_new_siberian_frontier" parliament issue).
country_event = {
	id = buff_colony.1
	title = buff_colony.1.t
	desc = buff_colony.1.d
	picture = COLONIZATION_eventPicture

	hidden = yes
	is_triggered_only = yes

	immediate = {
		event_target:buff_finished_colony = {
			every_neighbor_province = {
				limit = { is_empty = yes }
				add_siberian_construction = 400
			}
		}
	}

	option = {
		name = buff_colony.1.a
	}
}
