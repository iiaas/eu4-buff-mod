on_startup = {
	if = {
		limit = { ai = no }
		add_mercantilism = 100
		add_country_modifier = { name = luck duration = -1 hidden = yes }

		set_estate_privilege = estate_church_land_rights
		set_estate_privilege = estate_church_religious_state
		set_estate_privilege = estate_church_religious_diplomats
		set_estate_privilege = estate_church_development_of_temples

		set_estate_privilege = estate_nobles_nobility_primacy
		set_estate_privilege = estate_nobles_supremacy_over_crown
		set_estate_privilege = estate_nobles_development_of_castles
		set_estate_privilege = estate_nobles_officer_corp

		set_estate_privilege = estate_burghers_land_of_commerce
		set_estate_privilege = estate_burghers_financial_demand
		set_estate_privilege = estate_burghers_development_of_cities
		set_estate_privilege = estate_burghers_admirals
		
		change_estate_land_share = { estate = all share = -100 }
	}
}

# root = province, from = old owner
on_province_owner_change = {
	if = {
		limit = { owner = { ai = no } }
		owner = {
			save_event_target_as = buff_new_owner
			add_accepted_culture = root
		}
		if = {
			limit = { NOT = { is_core = event_target:buff_new_owner } }
			add_territorial_core = event_target:buff_new_owner
		}
		add_core = event_target:buff_new_owner
		set_local_autonomy = 0
		add_nationalism = -30
		add_devastation = -100
		clear_rebels = yes
	}
}

on_monthly_pulse = {
	if = { 
		limit = { ai = no }
		every_ally = {
			add_favors = { who = ROOT amount = -10 }
			add_opinion = { who = ROOT modifier = opinion_improved_relations multiplier = 10 }
		}
		every_subject_country = {
			add_opinion = { who = ROOT modifier = opinion_improved_relations multiplier = 10 }
		}
		every_war_enemy_country = {
			add_spy_network_from = { who = ROOT value = 100 }
		}
		every_province = {
			limit = {
				owner = { OR = { tag = ROOT is_subject_of = ROOT } }
				is_colony = no
			}
			every_empty_neighbor_province = { add_siberian_construction = 1 }
		}
	}
}

# root = location, from = winner country
on_battle_lost_province = {
	if = { 
		limit = { FROM = { ai = no } }
		FROM = {
			add_war_exhaustion = -1
			add_army_professionalism = 0.01
		}
	}
}
